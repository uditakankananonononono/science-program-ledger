"""Frozen R-04 driver (prereg 199 v2). env: R04_REPO (pinned repo dir), R04_OUT (output json). R04_DRYRUN=1 uses synthetic data only."""
import os, sys, json, numpy as np, pandas as pd
from sklearn.linear_model import LogisticRegression
from sklearn.preprocessing import StandardScaler
from sklearn.pipeline import make_pipeline
from sklearn.model_selection import StratifiedKFold
from sklearn.metrics import roc_auc_score
REPO=os.environ['R04_REPO']; OUT=os.environ['R04_OUT']; DRY=os.environ.get('R04_DRYRUN')=='1'
BASE=['fba_min','fba_rich','gnn','cnn','kmer']; LABELFREE=['fba_min','fba_rich']
if DRY:
    rng=np.random.default_rng(0); n=1249
    al=pd.DataFrame({'bnumber':[f'b{i:04d}' for i in range(n)],'essential':(rng.random(n)<0.17).astype(int),**{c:rng.random(n) for c in BASE}})
    oof=pd.DataFrame({'bnumber':al.bnumber,'essential':al.essential,'ensemble':rng.random(n)}); ger=al[['bnumber','essential']].copy()
else:
    al=pd.read_csv(f'{REPO}/results/aligned_predictions_all_models.csv'); oof=pd.read_csv(f'{REPO}/results/ensemble_oof.csv'); ger=pd.read_csv(f'{REPO}/data/gerdes_labels.csv')
y=al.essential.values.astype(int)
def stack(cols,seed,yv=None,X=None):
    yy=y if yv is None else yv; X=al[cols].values if X is None else X; out=np.zeros(len(yy))
    for tr,te in StratifiedKFold(3,shuffle=True,random_state=seed).split(X,yy):
        m=make_pipeline(StandardScaler(),LogisticRegression(max_iter=4000,class_weight='balanced',C=0.5)).fit(X[tr],yy[tr]); out[te]=m.predict_proba(X[te])[:,1]
    return out
R={}
# A1
oo=oof.merge(al[['bnumber','essential']],on='bnumber',suffixes=('_o','_a')); R['A1_n_join']=len(oo)
R['A1_oof_auroc']=float(roc_auc_score(oo.essential_a,oo.ensemble)); R['A1_oof_file_label_agree_with_aligned']=float((oo.essential_o==oo.essential_a).mean())
gd=ger.groupby('bnumber').essential.agg(lambda v: v.iloc[0] if v.nunique()==1 else np.nan).reset_index()  # duplicate bnumbers in gerdes_labels.csv: conflicting duplicates become NaN (= counted as disagreement)
R['A1_gerdes_rows']=int(len(ger)); R['A1_gerdes_conflicting_dup_bnumbers']=int(gd.essential.isna().sum())
j=al[['bnumber','essential']].merge(gd[['bnumber','essential']],on='bnumber',how='left',suffixes=('_a','_g'))
R['A1_missing_in_gerdes']=int(j.essential_g.isna().sum()); mm=j[j.essential_g.notna()&(j.essential_a!=j.essential_g)]
R['A1_label_mismatches']=int(len(mm)); R['A1_mismatch_bnumbers']=mm.bnumber.tolist(); R['A1_label_agree_frac']=float(1-(len(mm)+R['A1_missing_in_gerdes'])/len(al))
# A2
s7=stack(BASE,7); R['A2_auroc']=float(roc_auc_score(y,s7))
single={c:float(roc_auc_score(y,al[c].values)) for c in BASE}; R['single_auroc']=single
best=max(single,key=single.get); R['strongest_single']=best
R['fba_min_single_auroc_minus_claimed_0.666']=single['fba_min']-0.666  # flag only, not a gate
# B
Bs=[]
for s in range(1001,1021): Bs.append(float(roc_auc_score(y,stack(BASE,s))))
R['B_auroc_by_seed']=Bs; R['B_mean']=float(np.mean(Bs)); R['B_min']=float(np.min(Bs)); R['B_all_exceed_best_single']=bool(min(Bs)>single[best])
# C
rng=np.random.default_rng(7); d=[]
bc=al[best].values
while len(d)<2000:
    i=rng.integers(0,len(y),len(y))
    if y[i].min()==y[i].max(): continue
    d.append(roc_auc_score(y[i],s7[i])-roc_auc_score(y[i],bc[i]))
d=np.array(d); R['C_delta_mean']=float(d.mean()); R['C_ci95']=[float(np.percentile(d,2.5)),float(np.percentile(d,97.5))]
# D permutation (labels permuted before CV, scores fixed)
rng=np.random.default_rng(1); P=[]
for _ in range(200):
    yp=rng.permutation(y); P.append(float(roc_auc_score(yp,stack(BASE,7,yv=yp))))
R['D_perm_mean']=float(np.mean(P)); R['D_perm_p95']=float(np.percentile(P,95)); R['D_perm_max']=float(np.max(P)); R['D_real_above_max']=bool(R['A2_auroc']>max(P))
# D2 shuffled score columns
rng=np.random.default_rng(2); S=[]
for _ in range(50):
    X=al[BASE].values.copy()
    for k in range(X.shape[1]): X[:,k]=rng.permutation(X[:,k])
    S.append(float(roc_auc_score(y,stack(BASE,7,X=X))))
R['D2_shuffled_cols_mean']=float(np.mean(S))
# F label-free-only stack (report only)
F=[float(roc_auc_score(y,stack(LABELFREE,s))) for s in [7]+list(range(1001,1021))]
R['F_labelfree_stack_auroc_seed7']=F[0]; R['F_labelfree_by_seed_1001_1020']=F[1:]
sf=stack(LABELFREE,7); rng=np.random.default_rng(7); d=[]
bl=max(LABELFREE,key=lambda c:single[c]); R['F_best_labelfree_single']=bl
while len(d)<2000:
    i=rng.integers(0,len(y),len(y))
    if y[i].min()==y[i].max(): continue
    d.append(roc_auc_score(y[i],sf[i])-roc_auc_score(y[i],al[bl].values[i]))
R['F_delta_vs_best_labelfree_single']={'mean':float(np.mean(d)),'ci95':[float(np.percentile(d,2.5)),float(np.percentile(d,97.5))]}
json.dump(R,open(OUT,'w'),indent=1,default=float); print(json.dumps(R,indent=1,default=float))
