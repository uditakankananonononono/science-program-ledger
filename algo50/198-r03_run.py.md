"""Frozen R-03 driver (prereg 198). usage: python r03_run.py A | C
A: unmodified study12_redock.py run in a scratch copy of the pinned repo, then B/D scoring of its pose.
C: study12_redock.py logic with ETKDG seed = Vina seed = s for s in 196001..196010 (the ONLY change vs the repo script: seeds, scratch paths, scoring)."""
import sys, os, json, shutil, subprocess, pathlib, time, signal
import numpy as np
HERE=pathlib.Path(__file__).resolve().parent
SRC=pathlib.Path(os.environ['R03_REPO'])           # pinned repo checkout
WORK=pathlib.Path(os.environ['R03_WORK'])           # scratch
sys.path.insert(0,str(HERE))
from r03_metric import crystal_mol, pose_mol, inplace_rmsd, hungarian_inplace
TIMEOUT=3600
def fresh(tag):
    d=WORK/tag
    if d.exists(): shutil.rmtree(d)
    shutil.copytree(SRC,d); return d
def score(d, pose_text):
    ref=crystal_mol(str(d/'data/raw/7KX5.pdb'),str(d/'data/raw/X7V_ideal.sdf'))
    pm=pose_mol(pose_text)
    r,nm=inplace_rmsd(ref,pm)
    # D: same primary metric already symmetry+stereo aware; secondary graph-free Hungarian (squared distances, no superposition)
    rc=ref.GetConformer().GetPositions(); pc=pm.GetConformer().GetPositions()
    el_r=[a.GetSymbol() for a in ref.GetAtoms()]; el_p=[a.GetSymbol() for a in pm.GetAtoms()]
    return {'inplace_rmsd_primary':r,'n_mappings':nm,'hungarian_inplace_secondary':hungarian_inplace(rc,el_r,pc,el_p)}
def part_A():
    d=fresh('A'); t=time.time()
    p=subprocess.run([sys.executable,'studies/study12_redock.py'],cwd=d,env={**os.environ,'PYTHONPATH':str(d)},capture_output=True,text=True,timeout=TIMEOUT)
    out={'returncode':p.returncode,'secs':time.time()-t,'stdout_tail':p.stdout[-600:],'stderr_tail':p.stderr[-300:]}
    try:
        rep=json.loads((d/'results/redock_7KX5.json').read_text()); out['repo_reported']=rep
        out['scored']=score(d,(d/'data/raw/X7V_redock_pose.pdbqt').read_text())
    except Exception as e: out['error']=repr(e)
    json.dump(out,open(WORK/'A_result.json','w'),indent=1); print(json.dumps(out,indent=1))
def worker_C(s):
    """one seed, run inside its own subprocess (so the timeout can kill a hung native Vina call)"""
    sys.path.insert(0,str(SRC))
    from rdkit import Chem
    from drugdisc.prep import smiles_to_pdbqt
    from drugdisc.dock import dock_ligand
    d=fresh(f'C{s}'); raw=d/'data/raw'
    mol=Chem.MolFromMolFile(str(raw/'X7V_ideal.sdf'),removeHs=True); smi=Chem.MolToSmiles(mol)
    smiles_to_pdbqt(smi,str(raw/'X7V_ligand.pdbqt'),seed=s)
    box=json.loads((raw/'box_7kx5.json').read_text())
    result,pose=dock_ligand(str(raw/'7KX5_A_receptor.pdbqt'),str(raw/'X7V_ligand.pdbqt'),box['center'],box['size'],ligand_name='JUN8-76-3A',exhaustiveness=16,n_poses=9,cpu=2,seed=s)
    rec={'seed':s,'best_affinity':result.best_affinity,'scored':score(d,pose)}
    print('RESULT_JSON '+json.dumps(rec))
def part_C():
    res=[]
    for s in range(196001,196011):
        t=time.time(); rec={'seed':s}
        try:
            p=subprocess.run([sys.executable,str(HERE/'r03_run.py'),'Cseed',str(s)],capture_output=True,text=True,timeout=TIMEOUT,env=os.environ)
            line=[l for l in p.stdout.splitlines() if l.startswith('RESULT_JSON ')]
            if p.returncode!=0 or not line: raise RuntimeError('worker failed rc=%s %s'%(p.returncode,p.stderr[-300:]))
            rec=json.loads(line[0][len('RESULT_JSON '):])
        except Exception as e: rec['error']=repr(e); rec['failure']=True
        rec['secs']=time.time()-t; res.append(rec); print(json.dumps(rec),flush=True)
        json.dump(res,open(WORK/'C_result.json','w'),indent=1)
        shutil.rmtree(WORK/f'C{s}',ignore_errors=True)
    vals=[r['scored']['inplace_rmsd_primary'] if 'scored' in r else float('inf') for r in res]
    summ={'n_lt2':int(sum(v<2.0 for v in vals)),'median_all10_failures_as_inf':float(np.median(vals)),'rmsds':vals}
    json.dump({'runs':res,'summary':summ},open(WORK/'C_result.json','w'),indent=1); print(json.dumps(summ))
def selftest():
    """meeko-oracle direction proof on the PRE-docking ligand PDBQT; reference-only checks. No docked pose involved."""
    from meeko import PDBQTMolecule, RDKitMolCreate
    from r03_metric import crystal_mol, pose_mol, inplace_rmsd
    from rdkit import Chem
    txt=(SRC/'data/raw/X7V_ligand.pdbqt').read_text()
    mine=pose_mol(txt)
    om=RDKitMolCreate.from_pdbqt_mol(PDBQTMolecule(txt,skip_typing=True))[0]
    om=Chem.RemoveHs(om)
    mc=mine.GetConformer().GetPositions()
    mt=om.GetSubstructMatches(mine,uniquify=False,useChirality=True)
    oc=om.GetConformer().GetPositions()
    dev=min(float(np.sqrt(((oc[list(m)]-mc)**2).sum(1).mean())) for m in mt)
    ref=crystal_mol(str(SRC/'data/raw/7KX5.pdb'),str(SRC/'data/raw/X7V_ideal.sdf'))
    out={'meeko_oracle_max_dev_A':dev,'crystal_self_rmsd':inplace_rmsd(ref,ref)[0]}
    print(json.dumps(out)); assert dev<1e-3 and out['crystal_self_rmsd']<1e-6
if __name__=='__main__':
    if sys.argv[1]=='Cseed': worker_C(int(sys.argv[2]))
    else: {'A':part_A,'C':part_C,'selftest':selftest}[sys.argv[1]]()
