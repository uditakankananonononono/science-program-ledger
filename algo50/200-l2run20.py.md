# (v2: database split into 16 record-interleaved chunk files, one (query, chunk) task per worker process; results concatenated per query.)
# Wrapper: L2 with the COMMITTED prefilter (20), same alignment logic as the committed driver's _l2_worker (copied verbatim),
# but HSPs that cannot affect the hit rule (coverage < 0.8 or identity < 0.7) are dropped at collection (memory); stored values are unchanged.
import sys, json, os, multiprocessing as mp, parasail
sys.path.insert(0, '.')
import r05_driver as d
assert d.L2_PREFILTER_SCORE == 20
def w(args):
    qid, q, sprot_path = args
    mat = parasail.blosum62; prof = parasail.profile_create_16(q, mat)
    hsps = []
    with d.opn(sprot_path) as f:
        name, buf = None, []
        def handle(name, t):
            r = parasail.sw_striped_profile_16(prof, t, d.SW_OPEN, d.SW_EXT)
            if r.score >= d.L2_PREFILTER_SCORE:
                tr = parasail.sw_trace_striped_16(q, t, d.SW_OPEN, d.SW_EXT, mat)
                tb = tr.traceback; qa, ta = tb.query, tb.ref
                nq = sum(ch != '-' for ch in qa)
                qs = tr.end_query + 2 - nq
                qe = tr.end_query + 1
                idn, cov, al = d.hsp_stats(qa, ta, qs, qe, len(q))
                if cov >= 0.8 and idn >= 0.7: hsps.append(dict(subject=d.acc_of(name), identity=idn, coverage=cov, alnlen=al, score=tr.score))
        for l in f:
            if l.startswith('>'):
                if name is not None: handle(name, ''.join(buf))
                name, buf = l[1:].strip(), []
            else: buf.append(l.strip())
        if name is not None: handle(name, ''.join(buf))
    return qid, hsps
if __name__ == '__main__':
    qs = d.read_fasta_dir([sys.argv[1]]); out = sys.argv[2]
    os.makedirs(out, exist_ok=True)
    chunks = sorted(os.path.join('chunks', c) for c in os.listdir('chunks'))
    res = {n: [] for n, _ in qs}
    with mp.Pool(2, maxtasksperchild=1) as p:
        for qid, h in p.imap_unordered(w, [(n, s, c) for n, s in qs for c in chunks]): res[qid].extend(h)
    json.dump(dict(arm='L2', parasail_version='1.3.4', matrix='BLOSUM62', gap_open=d.SW_OPEN, gap_extend=d.SW_EXT, prefilter_score=20,
                   stored_filter='coverage>=0.8 and identity>=0.7', sprot_sha256=d.sha('sprot.fasta'), queries=[n for n, _ in qs], hsps=res),
              open(f"{out}/l2_hits.json", 'w'))
